== check.lua (runs inside the test nvim) ==
local out = {}
local function p(...) table.insert(out, table.concat(vim.tbl_map(tostring, { ... }), " ")) end
local plugin = require("lazy.core.config").plugins["annodiff.nvim"]
p("source:", plugin.url or plugin.dir, "branch:", plugin.branch or "-", "commit:",
  vim.trim(vim.fn.system({ "git", "-C", plugin.dir, "rev-parse", "--short", "HEAD" })))
p("loaded at startup:", plugin._.loaded ~= nil)
p(":Annodiff defined (lazy stub):", vim.fn.exists(":Annodiff") == 2, " :AnnodiffLayout:", vim.fn.exists(":AnnodiffLayout") == 2)
p("<leader>gvv mapped:", vim.fn.maparg(" gvv", "n") ~= "" or vim.fn.maparg("<leader>gvv", "n") ~= "")
p("annodiff module before use:", package.loaded["annodiff"] ~= nil)
vim.cmd("Annodiff HEAD")
p("loaded after :Annodiff:", plugin._.loaded ~= nil, "reason:", vim.inspect(plugin._.loaded and plugin._.loaded.cmd))
p("opts applied, layout =", require("annodiff.config").options.layout)
p("tabs:", #vim.api.nvim_list_tabpages())
local wins = {}
for _, w in ipairs(vim.api.nvim_tabpage_list_wins(0)) do
  local b = vim.api.nvim_win_get_buf(w)
  table.insert(wins, vim.fn.fnamemodify(vim.api.nvim_buf_get_name(b), ":t") .. "[" .. vim.bo[b].filetype .. "]")
end
p("review windows:", table.concat(wins, " | "))
p("buffer-local \\a in file window:", vim.fn.maparg("<localleader>a", "n") ~= "" )
local ok, err = pcall(vim.cmd, "help annodiff-install")
p("help annodiff-install:", ok and (vim.fn.expand("%:t") .. ":" .. vim.fn.line(".")) or err)
if ok then vim.cmd("close") end
vim.cmd("AnnodiffClose")
p("tabs after close:", #vim.api.nvim_list_tabpages())
p("doc/tags in checkout:", vim.uv.fs_stat(plugin.dir .. "/doc/tags") ~= nil,
  " git status:", vim.trim(vim.fn.system({ "git", "-C", plugin.dir, "status", "--porcelain" })) == "" and "clean" or "dirty")
vim.fn.writefile(out, vim.env.CHECK_OUT)
